Import-Module WebAdministration; $poolName = "WindowsTaskManagerPool"; $poolEnvironment = Get-WebConfiguration -PSPath "MACHINE/WEBROOT/APPHOST" -Filter "system.applicationHost/applicationPools/add[@name='$poolName']/environmentVariables"; $poolEnvironment.Collection | Where-Object { $_.name -in @("Authentication__Google__ClientId", "Authentication__Google__ClientSecret") } | Select-Object name, @{Name="ValueLength"; Expression={$_.value.Length}}    


## 設定


Import-Module WebAdministration; $poolName = "WindowsTaskManagerProdPool"; $filter = "system.applicationHost/applicationPools/add[@name='$poolName']/environmentVariables"; $values = @{"ASPNETCORE_ENVIRONMENT"="MYPROD"; "Authentication__Google__ClientId"=[Environment]::GetEnvironmentVariable("Authentication__Google__ClientId","Machine"); "Authentication__Google__ClientSecret"=[Environment]::GetEnvironmentVariable("Authentication__Google__ClientSecret","Machine")}; foreach ($entry in $values.GetEnumerator()) { $currentCollection = (Get-WebConfiguration -PSPath "MACHINE/WEBROOT/APPHOST" -Filter $filter).Collection; $existing = $currentCollection | Where-Object { $_.name -eq $entry.Key }; if ($null -eq $existing) { Add-WebConfigurationProperty -PSPath "MACHINE/WEBROOT/APPHOST" -Filter $filter -Name "." -Value @{name=$entry.Key; value=$entry.Value} } else { Set-WebConfigurationProperty -PSPath "MACHINE/WEBROOT/APPHOST" -Filter "$filter/add[@name='$($entry.Key)']" -Name "value" -Value $entry.Value } }
